function(hyp_test target name module source)
  add_executable(${target} "${source}")
  hyp_executable(${target})
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_sources(${target} PRIVATE Support/TestSupport.h)
  set_target_properties(${target} PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${PROJECT_BINARY_DIR}")
  target_link_libraries(${target} PRIVATE ${module})
  add_test(NAME ${name} COMMAND ${target})
  set_tests_properties(${name} PROPERTIES TIMEOUT 60)
endfunction()
