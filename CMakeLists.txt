cmake_minimum_required(VERSION 3.28)
project(Hyperion VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This first backend requires Windows x64 and MSVC. Other backends are not implemented yet.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(HYP_ENABLE_RENDERDOC "Build the optional RenderDoc capture plugin" OFF)
option(HYP_ENABLE_TRACY "Enable Tracy profiling and its network listener in Core-linked executables" OFF)
include(cmake/Dependencies.cmake)
include(CTest)

include(cmake/Modules.cmake)

add_subdirectory(Source/Runtime/Core)
add_subdirectory(Source/Runtime/Tasks)
add_subdirectory(Source/Runtime/IO)
add_subdirectory(Source/Runtime/Reflection)
add_subdirectory(Source/Runtime/Serialization)
add_subdirectory(Source/Runtime/AssetTypes)
add_subdirectory(Source/Runtime/Config)
add_subdirectory(Source/Runtime/Plugins)
add_subdirectory(Source/Runtime/Platform)
add_subdirectory(Source/Runtime/Math)
add_subdirectory(Source/Runtime/Materials)
add_subdirectory(Source/Runtime/Assets)
add_subdirectory(Source/Runtime/Scene)
add_subdirectory(Source/Runtime/AssetImport)
add_subdirectory(Source/Runtime/Shaders)
add_subdirectory(Source/Runtime/RHI)
add_subdirectory(Source/Runtime/Renderer)
add_subdirectory(Source/Runtime/Gui)
if(HYP_ENABLE_RENDERDOC)
  add_subdirectory(Source/Runtime/Capture)
  add_subdirectory(Source/Plugins/RenderDoc)
endif()
add_subdirectory(Source/Backends/D3D12)
add_subdirectory(Source/Plugins/Triangle)
add_subdirectory(Source/Plugins/DebugUI)
add_subdirectory(Source/Plugins/ModelViewer)
add_subdirectory(Source/Plugins/SceneViewer)
add_subdirectory(Source/Applications/AssetTool)
add_subdirectory(Source/Applications/Viewer)
include(cmake/NativeContent.cmake)
if(BUILD_TESTING)
  add_subdirectory(Source/Tests)
endif()
if(CMAKE_GENERATOR MATCHES "^Visual Studio")
  include(cmake/VisualStudio.cmake)
endif()
