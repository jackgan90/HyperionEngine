cmake_minimum_required(VERSION 3.24)
project(HyperionProfilingTools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(TracyRoot "${CMAKE_CURRENT_LIST_DIR}/../../out/deps/tracy")
if(NOT EXISTS "${TracyRoot}/capture/src/capture.cpp")
    message(FATAL_ERROR "Run tools/Bootstrap.py --only tracy first")
endif()
set(NO_STATISTICS OFF)
set(VENDOR_GUI OFF)
set(CPM_SOURCE_CACHE "${CMAKE_CURRENT_LIST_DIR}/../../out/profiling-deps" CACHE PATH "Headless tool dependencies")
include("${TracyRoot}/cmake/config.cmake")
include("${TracyRoot}/cmake/vendor.cmake")
include("${TracyRoot}/cmake/server.cmake")
file(READ "${CMAKE_CURRENT_LIST_DIR}/../../dependencies.lock.json" DependencyLock)
string(JSON TracyRevision GET "${DependencyLock}" tracy commit)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/GitRef.hpp" CONTENT "#pragma once\nnamespace tracy { static inline const char* GitRef = \"${TracyRevision}\"; }\n" @ONLY)
add_executable(tracy-capture "${TracyRoot}/capture/src/capture.cpp" "${TracyRoot}/capture/src/CaptureOutput.cpp")
add_executable(tracy-csvexport "${TracyRoot}/csvexport/src/csvexport.cpp")
foreach(Tool tracy-capture tracy-csvexport)
    target_include_directories(${Tool} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(${Tool} PRIVATE TracyServer TracyGetOpt)
endforeach()
